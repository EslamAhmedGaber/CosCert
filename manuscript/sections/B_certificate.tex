\section{The \CosCert{} certificate in detail}\label{app:certificate}

\paragraph{Inputs and arithmetic} $H$ is a decimal and enters as an exact rational. The Kummer values
$p_k+\mathrm ir_k={}_1F_1(b;b+1;\mathrm ik\pi)$, $k\le J$, are computed in ball arithmetic \citep{Joh17} at $128$ bits;
every Gram entry used in the head blocks is a ball, and the double-precision copies used for the tail blocks carry
entrywise rounding bounds proved in the standard model of floating-point arithmetic ($u=2^{-53}$). \Cref{tab:errsources}
lists every error source, its bound and how the bound is evaluated.

\paragraph{Factorisation} By the parity of \cref{prop:coords}, the covariance of $(Z,\Lambda_1/\pi,\dots,\Lambda_J/(J\pi))$
splits into an odd block $A_O$ and the block of $(Z,\text{even coordinates})$ with Schur structure
$(1,g^\top;g,A_E)$, so that, for $z=\vartheta+\mathrm it$ and $\zeta=2cz$,
\[\mathcal M(z)^{-2}=\det(I-2zK)=\det(I+\zeta A_O)\det(I+\zeta A_E)\,(1-2z\lambda_+)\,\rho(z)\qquad(\cref{lem:branch,lem:interlace}).\]
Each block is split into a head of its first $h$ coordinates and a tail, $A=(A_{hh},A_{ht};A_{th},A_{tt})$, and evaluated
with \cref{lem:logdet}, in which $\log\det(I+\zeta M)=\sum_i\log(1+\zeta\mu_i)$ over the eigenvalues $\mu_i$ of $M$, with
principal logarithms, and $\|\cdot\|_*$ is the trace norm.

\begin{lemma}[Head and tail]\label[lemma]{lem:logdet}
Let $A$ be real symmetric positive semi-definite, partitioned as above, $\operatorname{Re}\zeta\ge0$, $W=A_{th}^\top A_{th}$,
$S=A_{tt}-\zeta A_{th}(I+\zeta A_{hh})^{-1}A_{ht}$, $s_2=\|A_{tt}\|_2+|\zeta|\tr W$ and $s_F=\|A_{tt}\|_F+|\zeta|\tr W$. Let $Q$ be
invertible, $\ell\in[0,\infty)^h$, $F=Q^{-1}A_{hh}Q-\operatorname{diag}(\ell)$ and $\epsilon_F=|\zeta|\|F\|_F$, and suppose
$\epsilon_F<\frac12$ and $|\zeta|s_2<\frac12$. Then:
\emph{(a)} $\log\det(I+\zeta A)=\log\det(I+\zeta A_{hh})+\log\det(I+\zeta S)$ and
$\big|\log\det(I+\zeta A_{hh})-\sum_{i\le h}\log(1+\zeta\ell_i)\big|\le h\epsilon_F/(1-\epsilon_F)$;
\emph{(b)} for $u,v\in\mathbb C^h$,
\begin{gather*}
\Big|u^\top(I+\zeta A_{hh})^{-1}v-\sum_{j\le h}\frac{(Q^\top u)_j(Q^{-1}v)_j}{1+\zeta\ell_j}\Big|\le\frac{\epsilon_F\,\|Q^\top u\|\,\|Q^{-1}v\|}{1-\epsilon_F},\\
\Big|\tr\big((I+\zeta A_{hh})^{-1}W\big)-\sum_{j\le h}\frac{(Q^{-1}WQ)_{jj}}{1+\zeta\ell_j}\Big|\le\frac{\epsilon_F\,\|Q\|_F\|Q^{-1}\|_F\tr W}{1-\epsilon_F};
\end{gather*}
\emph{(c)} $\|S\|_2\le s_2$, $\|S\|_F\le s_F$, $\tr S=\tr A_{tt}-\zeta\tr\big((I+\zeta A_{hh})^{-1}W\big)$ and
\begin{gather*}\log\det(I+\zeta S)=\zeta\tr S-\tfrac12\zeta^2\|A_{tt}\|_F^2+E,\\
|E|\le\tfrac12|\zeta|^2\big(2|\zeta|\|A_{tt}\|_2\tr W+|\zeta|^2(\tr W)^2\big)+\frac{|\zeta|^3s_2s_F^2}{3(1-|\zeta|s_2)};\end{gather*}
\emph{(d)} for $g=(g_h,g_t)$ and $e=g_t-\zeta A_{th}(I+\zeta A_{hh})^{-1}g_h$, one has $\|e\|\le\|g_t\|+|\zeta|(\tr W)^{1/2}\|g_h\|$ and
\[g^\top(I+\zeta A)^{-1}g=g_h^\top(I+\zeta A_{hh})^{-1}g_h+e^\top e+E_g,\qquad|E_g|\le\frac{|\zeta|s_2\|e\|^2}{1-|\zeta|s_2}.\]
\end{lemma}

\begin{proof}
Since $A_{hh}$ is positive semi-definite, $\ell\ge0$ and $\operatorname{Re}\zeta\ge0$, the normal matrices $I+\zeta A_{hh}$ and
$I+\zeta D$, $D=\operatorname{diag}(\ell)$, have eigenvalues of modulus at least $1$, so their inverses have spectral norm at
most $1$. Put $X=\zeta(I+\zeta D)^{-1}F$; then $\|X\|_2\le\epsilon_F$, $(I+\zeta A_{hh})^{-1}=Q(I+X)^{-1}(I+\zeta D)^{-1}Q^{-1}$,
$(I+X)^{-1}=I-X(I+X)^{-1}$ and $\|X(I+X)^{-1}\|_2\le\epsilon_F/(1-\epsilon_F)$, which gives~(b) with $|\tr(MN)|\le\|M\|_2\|N\|_*$ and
$\|Q^{-1}WQ\|_*\le\|Q^{-1}\|_F\|Q\|_F\tr W$. \emph{(a)} The Schur complement gives $\det(I+\zeta A)=\det(I+\zeta A_{hh})\det(I+\zeta S)$,
and $\det(I+\zeta A_{hh})=\det(I+\zeta D)\det(I+X)$, where every eigenvalue $\mu$ of $X$ has $|\mu|\le\epsilon_F<\frac12$ and
$|\log(1+\mu)|\le|\mu|/(1-|\mu|)$. Both identities hold for the principal logarithms, not only modulo $2\pi\mathrm i$:
replacing $\zeta$ by $s\zeta$, $0\le s\le1$, preserves the hypotheses, every term is continuous in $s$, and all vanish at
$s=0$. \emph{(c)} With $P=A_{th}(I+\zeta A_{hh})^{-1}A_{ht}$, $\|P\|_F\le\|P\|_*\le\|A_{th}\|_F^2=\tr W$, which gives the norm bounds
and the trace. As $|\zeta|\|S\|_2<\frac12$, $\log\det(I+\zeta S)=\sum_{k\ge1}(-1)^{k+1}\zeta^k\tr(S^k)/k$, and Schur's inequality
gives $|\tr S^k|\le\|S\|_2^{k-2}\|S\|_F^2$ for $k\ge2$, which bounds the terms $k\ge3$ by the last term of the bound on $|E|$;
finally $\tr S^2=\|A_{tt}\|_F^2-2\zeta\tr(A_{tt}P)+\zeta^2\tr P^2$ with $|\tr(A_{tt}P)|\le\|A_{tt}\|_2\tr W$ and $|\tr P^2|\le(\tr W)^2$.
\emph{(d)} By the block inverse, $g^\top(I+\zeta A)^{-1}g=g_h^\top(I+\zeta A_{hh})^{-1}g_h+e^\top(I+\zeta S)^{-1}e$, and
$(I+\zeta S)^{-1}=I-\zeta S(I+\zeta S)^{-1}$ with $\|(I+\zeta S)^{-1}\|_2\le(1-|\zeta|s_2)^{-1}$.
\end{proof}

In the computation $Q$ is an approximate eigenvector matrix of the midpoint of $A_{hh}$, computed in floating point, $\ell$
are its floating-point eigenvalues (positive for every $H=0.50,\dots,0.99$; the code stops if one is not), and $F$ is formed
in ball arithmetic, so the similarity is exact and no orthogonality of $Q$ is required; $\operatorname{Re}\zeta=2c\vartheta>0$.
Part~(a) gives the head log-determinants, (b) and~(c) the tail, and (b) and~(d), with $g$ split in the same way, the
rank-one term $g^\top(I+\zeta A_E)^{-1}g$ of $U$ below. For the eigenvalue bounds of the truncation, Gershgorin's theorem is
applied to $\operatorname{diag}(\ell)+F$, which is exactly similar to $A_{hh}$ and has a real spectrum: a connected union of $k$
discs contains exactly $k$ eigenvalues, so sorted lower bounds follow cluster by cluster \citep[Theorem~6.1.1]{HJ13}; by
Cauchy's interlacing theorem they bound the leading eigenvalues of the full odd and even blocks from below, and the even
block, which is coupled to $Z$ by a rank-one term, loses its largest bound (\cref{lem:interlace}). In~(c),
$\|A_{tt}\|_2\le\min\big(\|A_{tt}\|_F,\ \cH(k_0\pi)^{-1-2H}\big)$ with $k_0$ the first tail mode (\cref{thm:eig}), and $\tr A_{tt}$,
$\|A_{tt}\|_F^2$ and $\tr W$ are summed with rounding bounds. The head size $h\in\{8,16,32,64,128\}$ is chosen per
quadrature node as the smallest one that meets both conditions of \cref{lem:logdet} and gives an enclosure radius below
$10^{-12}$.

\paragraph{Positive eigenvalue} Put $U(\zeta)=1-\zeta g^\top(I+\zeta A_E)^{-1}g$, so that the function of
\cref{lem:interlace} is $u(z)=U(2cz)$. Then $\lambda_+$ is the unique root in $(0,1)$ of $U(c/\lambda)=\lambda$, equivalently
$u(1/(2\lambda))=\lambda$, and it is enclosed by a certified sign change of $U(c/\lambda)-\lambda$ evaluated in balls. It is needed only for the factorisation used by the truncation bound: the branch itself is the
principal one for every tilt $\vartheta<\frac12$ (\cref{lem:branch}), and $\lambda_+\le\Var Z=1$ (\cref{lem:interlace}) shows
that the moment generating function exists on the whole strip. Production uses $\vartheta=\frac14$.

\paragraph{Inversion sums} Write $S_\Delta(x)$ for the infinite tilted midpoint sum of the proof of \cref{thm:certificate}
and $S_{\Delta,K}(x)$ for its first $K$ terms, which are the ones evaluated. The node spacing $\Delta$ is a binary number
chosen in floating point; the alias identity holds for the lattice actually summed, so every bound uses its exact period
$L=2\pi/\Delta$, enclosed in a ball. The upper alias sum is evaluated term by term in balls,
$P(Z^2>y)=\operatorname{erfc}(\sqrt{y/2})$, until a term falls below $2^{-200}$, and its remainder after $M$ terms is
bounded through $\operatorname{erfc}(v)\le e^{-v^2}$ by $e^{-x/2}r^{M+1}/(1-r)$, $r=e^{-(1/2-\vartheta)L}$. The truncation
bound is $\epsilon_K=\frac{2}{\pi|I|}e^{-\vartheta x}(1-2\vartheta)^{-1/2}\prod_{i\in I}(2s_Kc\,\nu_i)^{-1/2}$ with
$s_K=(K-\frac12)\Delta$, the certified eigenvalue lower bounds $\nu_i$ above and $I=\{i:2s_Kc\,\nu_i>1\}$, evaluated in
balls.

\paragraph{Procedure and parameters} $J=16384$, $t=25$, $\vartheta=\frac14$; $L$ is chosen so that the alias bounds are
below $10^{-14}$ and $K$ so that $\epsilon_K\le10^{-14}$; the head size $h\in\{8,\dots,128\}$ at each node is the smallest
that meets the conditions above and gives a radius below $10^{-12}$. These choices affect only the width of the
enclosure, never its validity. A critical value is obtained in two stages. First, a computed quantile $q$ is found by
Brent's method on the double-precision distribution function, together with an estimate $\rho$ of the relative width
needed. Second, the certified distribution function is evaluated at both ends of
$[q(1-f\rho-10^{-12}),\,q(1+f\rho+10^{-12})]$, starting with $f=1.5$ and doubling $f$ up to five times; the bracket is
accepted only if the upper end of the enclosure at the left point is below $p$ and the lower end at the right point
is above $p$. A value for which no bracket is accepted is reported as computed, not certified; this did not occur.
The printed critical value is the shortest decimal whose rounding changes the computed size by at most $2\times10^{-5}$,
and its size is certified separately.

\paragraph{Error budget} \Cref{tab:budget} lists the contributions at three values of $H$. At $H=\frac12$ the remainder
term dominates and decreases like $J^{-3/2}$; at larger $H$ it decreases faster, and the radius of the finite-dimensional
computation, which is independent of $J$, becomes the limiting term.

\paragraph{What the enclosure covers} Every error source of \cref{tab:errsources} is inside the interval; the
exceptional events of probability $e^{-t}$ enter as additive terms, so the enclosure is deterministic. Floating point
is used in two ways only. It selects parameters and approximate bases ($L$, $K$, $h$, $Q$, the computed quantile),
which can affect the width but not the validity. And the tail blocks are formed in double precision with error bounds
proved in the standard model (IEEE binary64, round to nearest; the bounds hold for any summation order). An earlier
version evaluated the alias and truncation bounds in double precision with a heuristic safety factor, stopped the alias
series without a remainder bound, rounded the evaluation points $cs_{\mathrm{lo}}$, $cs_{\mathrm{hi}}$ to nearest and
used a residual eigenvalue bound that does not guarantee sorted pairing; all of these steps are now in balls as
described above, and every certified value of $T_H$ in this paper was recomputed with the corrected code
(\cref{app:repro}). The correctness of the software (Arb, python-flint, NumPy and our code) is assumed, not proved; the
validation below and \cref{app:repro} are the checks on it.

\paragraph{Fixed-smoothing laws} The limits $\tau_{H,m}$ and $t_{H,B}$ in \eqref{eq:limits} are $Z/(w\sum_{i=1}^kV_i^2)^{1/2}$
with $(V_1,\dots,V_k)=(C_1,S_1,\dots,C_m,S_m)$, $k=2m$, $w=1/m$, or $(V_1,\dots,V_k)=(\Lambda_1,\dots,\Lambda_B)$, $k=B$, $w=1/B$;
let $\Sigma$ be the covariance of $(Z,V_1,\dots,V_k)$.

\begin{proposition}[Certified fixed-smoothing laws]\label[proposition]{prop:fixedcert}
Let $0<H<1$ and $c>0$. Then $P\big(Z^2\le cw\sum_iV_i^2\big)=F:=P\big(\theta_+Y_0^2\le\sum_{i=1}^ka_iY_i^2\big)$ for independent
standard normal $Y_0,\dots,Y_k$, where $\theta_+>0$ and $-a_1,\dots,-a_k<0$ are the eigenvalues of
$\Sigma^{1/2}\big(e_1e_1^\top-cw\operatorname{diag}(0,I_k)\big)\Sigma^{1/2}$. Let $\beta=\min_ia_i$, $\upsilon=\beta/(\theta_++\beta)$
and let $\varpi_j\ge0$ be the coefficients of $\prod_i(\beta/a_i)^{1/2}(1-\gamma_ix)^{-1/2}$, $\gamma_i=1-\beta/a_i\in[0,1)$, so
that $\sum_j\varpi_j=1$. Then $F=\sum_{j\ge0}\varpi_j\,I_\upsilon\big(\tfrac12,\tfrac k2+j\big)$, with $I_\upsilon$ the regularised
incomplete Beta function, and for every $K\ge0$, with $F_K=\sum_{j\le K}\varpi_jI_\upsilon(\frac12,\frac k2+j)$ and
$\bar\varpi_K=1-\sum_{j\le K}\varpi_j$,
\[F_K+\bar\varpi_K\,I_\upsilon\big(\tfrac12,\tfrac k2+K+1\big)\le F\le F_K+\bar\varpi_K.\]
$F$ decreases in $\theta_+$ and increases in each $a_i$; hence, if $\theta_+\in[\theta^-,\theta^+]$ and $a_i\in[a_i^-,a_i^+]$
with $a_i^->0$, the lower bound evaluated at $(\theta^+,a^-)$ and the upper bound evaluated at $(\theta^-,a^+)$ enclose~$F$.
\end{proposition}

\begin{proof}
$\Sigma$ is the Gram matrix of linearly independent functions in the inner product of $B_H$, hence positive definite, and
$e_1e_1^\top-cw\operatorname{diag}(0,I_k)$ has inertia $(1,k,0)$, which the congruence by $\Sigma^{1/2}$ preserves (Sylvester's
law of inertia); diagonalising gives the first identity. \Citet{Rub62} expands $\sum_ia_iY_i^2$ as a mixture of
$\beta\chi^2_{k+2j}$ laws with weights $\varpi_j$, and since $Y_0^2/(Y_0^2+\chi^2_\nu)$ has the Beta$(\frac12,\frac\nu2)$ law,
$P(\theta_+Y_0^2\le\beta\chi^2_{k+2j})=I_\upsilon(\frac12,\frac k2+j)$. As $I_\upsilon(\frac12,b)$ increases with $b$ and does not
exceed $1$, the omitted terms lie between $\bar\varpi_KI_\upsilon(\frac12,\frac k2+K+1)$ and $\bar\varpi_K$. Pathwise, the event
$\{\theta_+Y_0^2\le\sum_ia_iY_i^2\}$ grows with each $a_i$ and shrinks as $\theta_+$ grows.
\end{proof}

In the computation $\Sigma$ is assembled from Gram entries in ball arithmetic (\cref{prop:coords,prop:continuation}); the
weights, the eigenvalues of a $(k+1)$-dimensional matrix, are enclosed by an exact similarity transformation in ball
arithmetic followed by Gershgorin's theorem on clusters of discs; the $\varpi_j$ are computed in ball arithmetic as a power
series, and $I_\upsilon$ by $I_\upsilon(\frac12,b+1)=I_\upsilon(\frac12,b)+\upsilon^{1/2}(1-\upsilon)^b/\{b\,B(\frac12,b)\}$. Every one of the
\pthree{fx-n-cells} critical values was certified with the tightest bracket tried, $[q(1-10^{-9}),q(1+10^{-9})]$ for the
squared statistic.

\paragraph{Validation} At $H=\frac12$, where $T_{1/2}$ has a closed-form law \citep{AP97}, the certified quantiles at
$J=512$ and $4096$ enclose the exact values, and the computed ones agree with them to
\pthree{g0-min-digits}--\pthree{g0-max-digits} significant digits. The Cram\'er--von Mises quantiles $0.347305$ and
$0.461361$ of $\int\tilde W^2$ \citep{KPSS92} are reproduced by the same engine.
